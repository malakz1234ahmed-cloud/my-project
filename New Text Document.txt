<script>
    let isSwitchClosed = false;

    function switchTab(tabId) {
        document.querySelectorAll('.section').forEach(sec => sec.classList.remove('active'));
        document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
        
        document.getElementById(tabId).classList.add('active');
        event.target.classList.add('active');
    }

    function toggleSwitch() {
        isSwitchClosed = !isSwitchClosed;
        const btn = document.getElementById('switchBtn');
        const boardSwitch = document.getElementById('boardSwitch');

        if (isSwitchClosed) {
            btn.textContent = "المفتاح: مغلق (الدائرة متصلة)";
            btn.classList.add('closed');
            boardSwitch.innerHTML = "🔗 مفتاح مغلق";
        } else {
            btn.textContent = "المفتاح: مفتوح (الدائرة مفصولة)";
            btn.classList.remove('closed');
            boardSwitch.innerHTML = "🔌 مفتاح مفتوح";
        }
        updateCircuit();
    }

    function updateCircuit() {
        const volt = parseFloat(document.getElementById('voltSlider').value);
        const res = parseFloat(document.getElementById('resSlider').value);

        document.getElementById('voltVal').textContent = volt;
        document.getElementById('resVal').textContent = res;
        document.getElementById('boardVolt').textContent = volt + "V";
        document.getElementById('boardRes').textContent = res + "Ω";

        const bulb = document.getElementById('bulb');
        const bulbStatus = document.getElementById('bulbStatus');
        const wireBox = document.getElementById('wireBox');
        const currentDisplay = document.getElementById('currentVal');
        const powerDisplay = document.getElementById('powerVal');
        const banner = document.getElementById('feedbackBanner');

        if (!isSwitchClosed) {
            bulb.style.background = "#334155";
            bulb.style.boxShadow = "none";
            bulbStatus.textContent = "المصباح مطفأ";
            wireBox.classList.remove('active-flow');
            currentDisplay.textContent = "0.00";
            powerDisplay.textContent = "0.00";
            banner.textContent = "الدائرة مفتوحة: لا يوجد مسار مغلق لمرور الإلكترونات.";
            banner.style.color = "#cbd5e1";
            return;
        }

        const current = volt / res;
        const power = volt * current;

        currentDisplay.textContent = current.toFixed(2);
        powerDisplay.textContent = power.toFixed(2);
        wireBox.classList.add('active-flow');

        if (current > 4.5) {
            bulb.style.background = "#ef4444";
            bulb.style.boxShadow = "0 0 35px #ef4444";
            bulbStatus.textContent = "⚠️ تحذير: تيار مرتفع جداً!";
            banner.textContent = "تنبيه: شدة التيار عالية جداً مما قد يؤدي إلى تلف المصباح! ارفع قيمة المقاومة.";
            banner.style.color = "#fca5a5";
        } else {
            const intensity = Math.min(current / 3, 1);
            bulb.style.background = `rgba(250, 204, 21, ${0.3 + intensity * 0.7})`;
            bulb.style.boxShadow = `0 0 ${15 + intensity * 40}px rgba(250, 204, 21, ${intensity})`;
            bulbStatus.textContent = current > 1.5 ? "إضاءة قوية" : "إضاءة متوسطة / خافتة";
            banner.textContent = `التيار يسري بنجاح وفق قانون أوم: (${volt}V ÷ ${res}Ω = ${current.toFixed(2)}A).`;
            banner.style.color = "#86efac";
        }
    }

    function checkQuiz() {
        const q1 = document.querySelector('input[name="q1"]:checked');
        const q2 = document.querySelector('input[name="q2"]:checked');
        const resultDiv = document.getElementById('quiz-result');

        if (!q1 || !q2) {
            resultDiv.textContent = "يرجى الإجابة على جميع الأسئلة أولاً.";
            resultDiv.style.color = "#f59e0b";
            return;
        }

        let score = 0;
        if (q1.value === 'b') score++;
        if (q2.value === 'a') score++;

        if (score === 2) {
            resultDiv.textContent = "🎉 ممتاز! إجاباتك صحيحة بنسبة 100% (2 من 2).";
            resultDiv.style.color = "#10b981";
        } else {
            resultDiv.textContent = `حصلت على ${score} من 2. راجع التجربة في المختبر التفاعلي وحاول مرة أخرى!`;
            resultDiv.style.color = "#ef4444";
        }
    }
